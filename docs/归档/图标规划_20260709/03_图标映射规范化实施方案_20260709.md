### 2.3 新增辅助函数

```typescript
/**
 * 获取图标的标准名（如果图标有 primaryName，返回 primaryName，否则返回 name）
 */
export const getIconStandardName = (iconName: string): string | null => {
    const config = ICON_CONFIG.find(c => c.name === iconName);
    if (!config) return null;
    return config.primaryName || config.name;
};

/**
 * 根据场景分类获取推荐图标列表
 */
export const getIconsByScene = (scene: string): IconConfig[] => {
    return ICON_CONFIG.filter(icon => icon.scene === scene);
};

/**
 * 校验图标名是否是标准名（不是别名）
 */
export const isStandardName = (iconName: string): boolean => {
    const config = ICON_CONFIG.find(c => c.name === iconName);
    return config ? !config.primaryName : false;
};
```

// 3. 导出函数：覆盖表 > 规则匹配 > 默认
export const getDocumentIcon = (documentTitle: string): string => {
    // 优先级1：手动覆盖
    if (OVERRIDE_MAPPING[documentTitle]) {
        return OVERRIDE_MAPPING[documentTitle];
    }
    
    // 优先级2：关键词规则匹配
    for (const rule of KEYWORD_RULES) {
        if (rule.pattern.test(documentTitle)) {
            return rule.icon;
        }
    }
    
    // 优先级3：默认降级
    return 'file-text';
};

// 保留旧接口以兼容
export const DOCUMENT_ICON_MAPPING = { ...OVERRIDE_MAPPING };

### 5.2 行为对比

| 场景 | 改造前 | 改造后 |
|------|--------|--------|
| 手动映射的 44 个文档 | 返回指定图标 | 返回指定图标（不变） |
| 未映射但包含关键词的新文档 | 返回 `file` | 自动匹配规则返回合适图标 |

### 5.3 改造后的维护指南

- **新增覆盖条目**：在 `OVERRIDE_MAPPING` 中添加 `"文档名.md": "图标名"`
- **新增规则**：在 `KEYWORD_RULES` 中添加 `{ pattern: /关键词1\|关键词2/i, icon: '图标名' }`
- **删除规则**：直接删除对应的规则条目
